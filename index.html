Бұл жобаның негізгі мақсаты — Arduino микроконтроллері негізінде автоматтандырылған басқару жүйесін әзірлеу және оның жұмыс принципін тәжірибе жүзінде көрсету.   1. Arduino Uno микроконтроллерінің жұмыс принципін зерттеу
  2. RFID жүйесінің жұмыс принципін қарастыру
  3. Сервопривод арқылы шлагбаумды басқару
  4. Жарық датчигінің көмегімен жарық деңгейін анықтау

БАҒДАРЛАМАЛЫҚ ҚАМТАМАСЫЗ ЕТУ

#include <SPI.h>
#include <MFRC522.h>
#include <Servo.h>

#define SS_PIN 10
#define RST_PIN 9
#define LDR_PIN A0
#define LED_PIN 3

MFRC522 rfid(SS_PIN, RST_PIN);
Servo barrier;
byte allowedUID[4] = {0xDE, 0xAD, 0xBE, 0xEF}; // мысал UID

void setup() {
  Serial.begin(9600);
  SPI.begin();
  rfid.PCD_Init();
  barrier.attach(6);
  barrier.write(0);
  pinMode(LED_PIN, OUTPUT);
}

void loop() {
  int light = analogRead(LDR_PIN);
  digitalWrite(LED_PIN, light < 400 ? HIGH : LOW);

  if (rfid.PICC_IsNewCardPresent() && rfid.PICC_ReadCardSerial()) {
    bool ok = true;
    for (byte i = 0; i < 4; i++) {
      if (rfid.uid.uidByte[i] != allowedUID[i]) ok = false;
    }
    if (ok) {
      Serial.println("Карта рұқсат етілді");
      barrier.write(90);
      delay(3000);
      barrier.write(0);
    } else {
      Serial.println("Карта қабылданбады");
    }
    rfid.PICC_HaltA();
  }
  delay(200);
}
